cmake_minimum_required(VERSION 3.22.1)
project(detector LANGUAGES CXX ASM)
add_library(detector SHARED detector.cpp)
target_link_libraries(detector dl)
target_compile_options(detector PRIVATE -Wall -Wextra -Werror)
add_library(probe SHARED probe.cpp clone-probe.S)
add_library(stateprobe SHARED stateprobe.cpp)
target_link_libraries(probe PRIVATE log dl)
target_link_libraries(stateprobe PRIVATE log dl)
foreach(layer detector probe stateprobe)
    target_link_options(${layer} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
