cmake_minimum_required(VERSION 3.22.1)
project(detector LANGUAGES CXX ASM)
add_library(detector SHARED detector.cpp)
target_link_libraries(detector dl)
target_compile_options(detector PRIVATE -Wall -Wextra -Werror)
add_library(probe SHARED probe.cpp clone-probe.S)
add_library(stateprobe SHARED stateprobe.cpp)
add_library(machineprobe SHARED machineprobe.cpp)
add_library(environmentprobe SHARED environmentprobe.cpp)
target_link_libraries(environmentprobe PRIVATE dl)
target_compile_options(environmentprobe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(probe PRIVATE log dl)
target_link_libraries(stateprobe PRIVATE log dl)
target_include_directories(stateprobe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../shared/native")
foreach(layer detector probe stateprobe machineprobe environmentprobe)
    target_link_options(${layer} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_library(incrementalprobe SHARED ../../debug/cpp/incrementalprobe.cpp)
    target_link_libraries(incrementalprobe PRIVATE dl)
    target_link_options(incrementalprobe PRIVATE "-Wl,-z,max-page-size=16384")
    add_library(capabilityprobe SHARED ../../debug/cpp/capabilityprobe.cpp)
    target_compile_options(capabilityprobe PRIVATE -Wall -Wextra)
    target_link_options(capabilityprobe PRIVATE "-Wl,-z,max-page-size=16384")
endif()
