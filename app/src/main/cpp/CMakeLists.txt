cmake_minimum_required(VERSION 3.22.1)
project(shield LANGUAGES CXX ASM)
add_library(shield SHARED shield.cpp syscall-dispatch.S)
target_compile_features(shield PRIVATE cxx_std_17)
target_compile_options(shield PRIVATE -fvisibility=hidden -fno-omit-frame-pointer)
target_link_libraries(shield PRIVATE log dl)
target_link_options(shield PRIVATE "-Wl,-z,max-page-size=16384")
foreach(layer vpnstate developerstate)
    add_library(${layer} SHARED ${layer}.cpp)
    target_compile_features(${layer} PRIVATE cxx_std_17)
    target_compile_options(${layer} PRIVATE -fvisibility=hidden -fno-omit-frame-pointer)
    target_link_libraries(${layer} PRIVATE log dl)
    target_link_options(${layer} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
